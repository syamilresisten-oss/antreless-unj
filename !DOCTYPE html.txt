        function openItemModal(umkmId, menuId) {
            const umkm = state.umkms.find(u => u.id === umkmId);
            if (!umkm) return;
            const menu = umkm.menus.find(m => m.id === menuId);
            if (!menu) return;

            tempSelectedItem = {
                umkmId: umkm.id,
                umkmName: umkm.name,
                canteenId: umkm.canteenId,
                menu: menu,
                quantity: 1
            };

            document.getElementById('modal-item-title').innerText = menu.name;
            document.getElementById('modal-item-desc').innerText = menu.desc || 'Makanan siap disajikan hangat.';
            document.getElementById('modal-item-price').innerText = `Rp ${menu.price.toLocaleString('id-ID')}`;
            document.getElementById('modal-item-qty').innerText = '1';
            document.getElementById('modal-item-note').value = '';
            
            updateModalTotal();
            openModal('modal-item-detail');
        }

        function changeQty(delta) {
            if (!tempSelectedItem) return;
            tempSelectedItem.quantity = Math.max(1, tempSelectedItem.quantity + delta);
            document.getElementById('modal-item-qty').innerText = tempSelectedItem.quantity;
            updateModalTotal();
        }

        function updateModalTotal() {
            if (!tempSelectedItem) return;
            const total = tempSelectedItem.menu.price * tempSelectedItem.quantity;
            document.getElementById('modal-item-total').innerText = `Rp ${total.toLocaleString('id-ID')}`;
        }

        function confirmAddToCart() {
            if (!tempSelectedItem) return;
            const note = document.getElementById('modal-item-note').value.trim();

            // If cart has item from different canteen, alert option to reset
            if (state.cart.length > 0 && state.cart[0].canteenId !== tempSelectedItem.canteenId) {
                if (!confirm("Keranjangmu berisi pesanan dari kantin lain. Kosongkan keranjang untuk mengganti kantin?")) {
                    return;
                }
                state.cart = [];
            }

            state.cart.push({
                ...tempSelectedItem,
                note: note
            });

            closeModal('modal-item-detail');
            updateCartBadge();
            alert(`✅ ${tempSelectedItem.menu.name} telah ditambahkan ke keranjang!`);
        }

        function updateCartBadge() {
            const totalItems = state.cart.reduce((sum, item) => sum + item.quantity, 0);
            document.getElementById('cart-count-badge').innerText = totalItems;
        }

        function calculateJastipFee() {
            const select = document.getElementById('jastip-target-building');
            const distanceMeter = parseInt(select.value) || 350;
            const fee = distanceMeter * 10; // Rp 10 / meter
            document.getElementById('calculated-jastip-fee').innerText = `Rp ${fee.toLocaleString('id-ID')}`;
            calculateCheckoutTotals();
            return fee;
        }

        function calculateCheckoutTotals() {
            const subtotal = state.cart.reduce((sum, item) => sum + (item.menu.price * item.quantity), 0);
            const isJastip = document.querySelector('input[name="fulfillment"]:checked').value === 'jastip';
            
            // Biaya Layanan Web / Aplikasi 5% dari subtotal makanan
            const appFee = Math.round(subtotal * 0.05);

            let deliveryFee = 0;
            if (isJastip) {
                const distanceMeter = parseInt(document.getElementById('jastip-target-building').value) || 350;
                deliveryFee = distanceMeter * 10;
            }

            const total = subtotal + appFee + deliveryFee;

            document.getElementById('checkout-subtotal').innerText = `Rp ${subtotal.toLocaleString('id-ID')}`;
            document.getElementById('checkout-app-fee').innerText = `Rp ${appFee.toLocaleString('id-ID')}`;
            document.getElementById('checkout-delivery-fee').innerText = `Rp ${deliveryFee.toLocaleString('id-ID')}`;
            document.getElementById('checkout-total').innerText = `Rp ${total.toLocaleString('id-ID')}`;
        }

           4. CHECKOUT & REAL-TIME ORDER LIFECYCLE
           ========================================================================== */

        function processCheckout() {
            if (state.cart.length === 0) {
                alert("Keranjang makananmu masih kosong!");
                return;
            }

            const name = document.getElementById('buyer-checkout-name').value.trim();
            const nim = document.getElementById('buyer-checkout-nim').value.trim();
            const nimError = document.getElementById('checkout-nim-error');

            // 10-Digit NIM Validation Rule
            const nimRegex = /^\d{10}$/;
            if (!name || !nimRegex.test(nim)) {
                nimError.classList.remove('hidden');
                alert("Mohon lengkapi Nama dan NIM (harus tepat 10 angka)!");
                return;
            }
            nimError.classList.add('hidden');

            const mode = document.querySelector('input[name="fulfillment"]:checked').value;
            const selectBuilding = document.getElementById('jastip-target-building');
            const selectedBuildingText = selectBuilding.options[selectBuilding.selectedIndex].text;
            const distanceMeter = parseInt(selectBuilding.value);
            const jastipFee = mode === 'jastip' ? distanceMeter * 10 : 0;
            const subtotal = state.cart.reduce((sum, item) => sum + (item.menu.price * item.quantity), 0);
            
            // Perhitungan Biaya Layanan Aplikasi 5%
            const appFee = Math.round(subtotal * 0.05);

            const newOrder = {
                orderId: 'UNJ-' + Math.floor(100000 + Math.random() * 900000),
                createdAt: new Date().toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' }),
                customerName: name,
                customerNim: nim,
                canteenId: state.cart[0].canteenId,
                umkmId: state.cart[0].umkmId,
                umkmName: state.cart[0].umkmName,
                items: [...state.cart],
                subtotal: subtotal,
                appFee: appFee, // Fee 5% untuk platform
                jastipFee: jastipFee,
                totalAmount: subtotal + appFee + jastipFee,
                mode: mode, // 'pickup' or 'jastip'
                targetLocation: mode === 'jastip' ? selectedBuildingText : 'Ambil Sendiri di Kantin',
                distanceMeter: mode === 'jastip' ? distanceMeter : 0,
                status: 'MENUNGGU_KONFIRMASI', // MENUNGGU_KONFIRMASI -> DIPROSES -> SIAP -> DALAM_PENGANTARAN -> SELESAI
                jastipCourierName: null,
                jastipCourierNim: null,
                rating: 0,
                review: ''
            };

            state.orders.unshift(newOrder);
            state.cart = [];
            updateCartBadge();
            saveState();

            closeModal('modal-cart');
            alert(`🎉 Pesanan ${newOrder.orderId} Berhasil Dibuat!\nStatus: Menunggu konfirmasi kedai.`);
            
            // Open Tracker Modal for buyer
            openOrderTrackerModal(newOrder.orderId);
        }

        function setOrderRating(orderId, rating) {
            const order = state.orders.find(o => o.orderId === orderId);
            if (!order) return;
            order.rating = rating;
            saveState();
            openOrderTrackerModal(orderId);

        function checkBuyerActiveOrder() {
            const banner = document.getElementById('buyer-active-order-banner');
            const statusText = document.getElementById('active-order-status-text');
            if (state.orders.length > 0) {
                const latest = state.orders[0];
                if (latest.status !== 'SELESAI') {
                    banner.classList.remove('hidden');
                    statusText.innerText = `Status Order: ${latest.status.replace(/_/g, ' ')}`;
                    return;
                }
            }
            banner.classList.add('hidden');
        }

                name: name,
                category: category,
                desc: desc,
                menus: [

                ]

            saveState();
            closeModal('modal-register-umkm');
            alert(`🎉 Kedai UMKM "${name}" berhasil didaftarkan!`);

        function saveNewMenu(e) {
            e.preventDefault();
            const name = document.getElementById('add-menu-name').value.trim();
            const price = parseInt(document.getElementById('add-menu-price').value) || 10000;
            const desc = document.getElementById('add-menu-desc').value.trim();
            const emoji = document.getElementById('add-menu-emoji').value.trim() || '🍱';

            if (state.umkms.length > 0) {
                state.umkms[0].menus.push({
                    id: 'm-' + Date.now(),
                    name: name,
                    price: price,
                    desc: desc,
                    emoji: emoji
                });
                saveState();
                renderStallsView();
                renderUMKMDashboard();
            }

            closeModal('modal-add-menu');
            alert(`✅ Menu "${name}" berhasil ditambahkan!`);
        }
